======
RULES
======

[ > Player | Crate ] -> [ > Player | > Crate ]

(Laser movement)
[ stationary LaserLeft ] -> [ left LaserLeft ]
[ stationary LaserRight ] -> [ right LaserRight ]
[ stationary LaserUp ] -> [ up LaserUp ]
[ stationary LaserDown ] -> [ Down LaserDown ]

late [Player LaserLeft] -> restart
late [crate LaserLeft] -> [Crate]
late [Wall LaserLeft] -> [Wall]
late [ LaserSpawnLeft no LaserLeft ] -> [ LaserSpawnLeft  LaserLeft ]
late [ MirrorA  LaserLeft ] -> [ MirrorA  LaserDown]
(late [ MirrorA  LaserRight ] -> [ MirrorA  LaserUp]
late [ MirrorA  LaserDown ] -> [ MirrorA  LaserRight]
late [ MirrorA  Laserup ] -> [ MirrorA  LaserLeft])



late Right [ BeamR | no Object ] -> [ BeamR | BeamLR ]
+late Right[ no Object | BeamL ] -> [ BeamRL | BeamL ]
+late Down [ BeamD | no Object ] -> [ BeamD | BeamUD ]
+late Down [ no Object | BeamU ] -> [ BeamDU | BeamU ]

+late Right[ BeamR | MirrorTR ] -> [ BeamR | MirrorTR BeamLD ]
+late Down [ MirrorTR | BeamU ] -> [ MirrorTR BeamDL | BeamU ]
+late Right[ BeamR | MirrorBR ] -> [ BeamR | MirrorBR BeamLU ]
+late Down [ BeamD | MirrorBR ] -> [ BeamD | MirrorBR BeamUL ]
+late Right[ MirrorBL | BeamL ] -> [ MirrorBL BeamRU | BeamL ]
+late Down [ BeamD | MirrorBL ] -> [ BeamD | MirrorBL BeamUR ]
+late Right[ MirrorTL | BeamL ] -> [ MirrorTL BeamRD | BeamL ]
+late Down [ MirrorTL | BeamU ] -> [ MirrorTL BeamDR | BeamU ]
